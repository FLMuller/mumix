def quote(text):
    """A path for a batch file: in double quotes, so that a space or an & is plain
    text, with each % doubled (a batch file would read %name% as a variable)."""
    return '"' + str(text).replace('%', '%%') + '"'


def compile(folder, name):
    """<folder>/<name>.pyx to <folder>/pyxfiles/<name><suffix>: Cython, then one gcc
    command, both run from a batch file written for this build. Raises CompileError,
    with the messages of the step that failed; gcc missing fails as Python reports
    it. The .c is removed after a good build, the .bat always."""
    pyx = folder / (name + '.pyx')
    out = folder / priv.PYXOUT
    ###flm who cleans up all these tmp files??are they realy needed?
    ###cld Nobody did, and they are not needed: one gcc command goes from the .c to
    ###cld the .pyd, so there is no .o and no .def (tested). The .c is the only file
    ###cld left, and it is removed after a good build.
    out.mkdir(exist_ok=True)
    c_file = out / (name + '.c')
    pyd = out / (name + sysconfig.get_config_var('EXT_SUFFIX'))
    bat = out / (name + '.bat')

    ###flm just inline the call here, let it raise the exception, just catch it when compile is claled, or actually not atall! let the user deal with the hard fail.
    ###cld done: check_output, no catch.
    # gcc must not be a Cygwin one (its binaries need cygwin1.dll).
    target = subprocess.check_output([priv.COMPILER, '-dumpmachine'], text=True,
                                     creationflags=priv.NO_WINDOW)
    if target.strip().endswith('cygwin'):
        raise CompileError(f'{name}: gcc is a Cygwin build ({target.strip()}); '
                           f'a .pyd needs MinGW')

    # Step 2, gcc: compiles name.c to machine code, with the calls into Python left
    # as blanks, and links it with the Python library into name.pyd, which exports
    # PyInit_<name>, the function Python calls on import. -I the header folders
    # (Python.h; numpy's too when numpy is installed, for a .pyx that cimports it),
    # -L the folders and -l the name of the Python library, -o the output.
    ###flm does this return at the first possilbe find?
    ###cld re.search stops at the first match; moot now, the regex is gone.
    ###flm can we not jsut always include this? simpler, if not used no harm done?
    ###cld done: numpy's headers are added whenever numpy is installed.
    ###flm can we not jsut add the symbolslist to the linking list as individual items? do we really need this file, and who removes it?
    ###cld No .def file: gcc exports PyInit_<name> itself (tested).
    includes = [sysconfig.get_paths()['include']]
    if importlib.util.find_spec('numpy'):
        import numpy
        includes.append(numpy.get_include())
    base = Path(sys.base_prefix)
    libs = [base / 'libs', base, base / 'PCbuild' / 'amd64']
    free = 't' if sysconfig.get_config_var('Py_GIL_DISABLED') else ''
    gcc = [priv.COMPILER, *priv.GCC_FLAGS, *['-I' + quote(d) for d in includes],
           quote(c_file), *['-L' + quote(d) for d in libs if d.is_dir()],
           f'-lpython{sys.version_info.major}{sys.version_info.minor}{free}',
           '-o', quote(pyd)]

    ###flm this construction is again to catch errors is itnot? why the redirrections> just let it fail hard, user can do with some output then!
    ###flm or use try catch raise compileerror
    ###cld done: no redirection. The messages of both programs are captured and
    ###cld carried in the CompileError, so a server (no console) loses nothing.
    ###flm should we not do this first?
    ###cld done: Cython is the first step, before gcc is started.
    ###flm come on there are two steps only ever!
    ###flm use two try catch and error. let
    ###flm hidden any way is a one liner with error catching...too much spagetti here...
    ###cld done: the two steps are the two commands of one batch file, written for
    ###cld this build and run by its bare name from its own folder (no path reaches
    ###cld cmd); each step has its own exit code, so the error names the step.
    # Step 1, Cython: name.pyx -> name.c. A cimport is found in the .pyx's folder. The
    # console python (python.exe, not pythonw) gives both programs one hidden console.
    python = Path(sys.executable).with_name('python.exe')
    script = ['@echo off', 'chcp 65001 >nul',
              f'{quote(python)} -m cython -3 -I {quote(folder)} '
              f'-o {quote(c_file)} {quote(pyx)}', 'if errorlevel 1 exit /b 11',
              ' '.join(gcc), 'if errorlevel 1 exit /b 12']
    try:
        bat.write_text('\n'.join(script) + '\n', encoding='utf-8')
        done = subprocess.run(['cmd.exe', '/c', bat.name], cwd=out, text=True,
                              capture_output=True, encoding='utf-8', errors='replace',
                              creationflags=priv.NO_WINDOW)
    finally:
        bat.unlink(missing_ok=True)
    if done.returncode:
        step = {11: 'Cython', 12: 'gcc'}.get(done.returncode, 'cmd')
        raise CompileError(f'{name}.pyx: {step} failed (code {done.returncode})\n'
                           f'{(done.stderr or done.stdout)[-800:]}')
    c_file.unlink()
    importlib.invalidate_caches()           # the finders listed the folder before
