def load(*items):
    """use() each item, build its .pyx if stale, and import its modules. Binds
    nothing and returns nothing; the first failure raises. A .pyx is built into
    <folder>/pyxfiles/<name><suffix>: Cython, then one gcc command, both run from a
    batch file written for this build. A failed build is a CompileError (an
    ImportError) with the messages of the step that failed. The .c is removed after
    a good build, the .bat always."""
    for folder, names in units(items):
        for name in names:
            pyx = folder / (name + '.pyx')
            rebuild = False
            if pyx.is_file():
                # Stale: no build for THIS Python, or it is older than the .pyx or
                # than a file it is built from (its own .pxd, the .pxd of each
                # module it cimports, the files it includes).
                needs = [pyx, folder / (name + '.pxd')]
                text = pyx.read_text(encoding='utf-8', errors='replace')
                for found in priv.NEEDS.finditer(text):
                    for m in (found.group(2) or found.group(1) or '').split(','):
                        if m.strip():
                            m = m.split(' as ')[0].strip().replace('.', os.sep)
                            needs.append(folder / (m + '.pxd'))
                    needs += [folder / found.group(3)] if found.group(3) else []
                suffix = sysconfig.get_config_var('EXT_SUFFIX')
                built = folder / priv.PYXOUT / (name + suffix)
                newest = max(p.stat().st_mtime for p in needs if p.is_file())
                rebuild = not built.is_file() or built.stat().st_mtime < newest
                if rebuild and name in sys.modules:
                    raise ImportError(
                        f'use: {name}.pyx is newer than the build this process '
                        f'imported: run again, to import the new build')
                addpath(built.parent)
            if rebuild:
                out = built.parent
                out.mkdir(exist_ok=True)
                c_file = out / (name + '.c')
                bat = out / 'tmpcompile.bat'

                # gcc must be there, and not a Cygwin one (its binaries need
                # cygwin1.dll).
                try:
                    target = subprocess.check_output(
                        [priv.COMPILER, '-dumpmachine'], text=True,
                        creationflags=priv.NO_WINDOW)
                except OSError as error:
                    raise CompileError(f'{name}: gcc was not found ({error})') from error
                if target.strip().endswith('cygwin'):
                    raise CompileError(f'{name}: gcc is a Cygwin build '
                                       f'({target.strip()}); a .pyd needs MinGW')

                # The gcc command: it compiles name.c to machine code, with the calls
                # into Python left as blanks, and links it with the Python library
                # into name.pyd, which exports PyInit_<name>, the function Python
                # calls on import. -I the header folders (Python.h; numpy's too when
                # numpy is installed, for a .pyx that cimports it), -L the folders and
                # -l the name of the Python library, -o the output. No .o and no .def
                # file are needed.
                includes = [sysconfig.get_paths()['include']]
                if importlib.util.find_spec('numpy'):
                    import numpy
                    includes.append(numpy.get_include())
                base = Path(sys.base_prefix)
                libs = [base / 'libs', base, base / 'PCbuild' / 'amd64']
                free = 't' if sysconfig.get_config_var('Py_GIL_DISABLED') else ''
                version = f'{sys.version_info.major}{sys.version_info.minor}{free}'
                gcc = [priv.COMPILER, *priv.GCC_FLAGS, *['-I' + str(d) for d in includes],
                       str(c_file), *['-L' + str(d) for d in libs if d.is_dir()],
                       f'-lpython{version}', '-o', str(built)]

                # Cython: name.pyx -> name.c. A cimport is found in the .pyx's folder.
                # The console python (python.exe, not pythonw) gives both programs
                # one hidden console. In the batch file every argument is in double
                # quotes, so that a space or an & in a path is plain text, and each %
                # is doubled (a batch file would read %name% as a variable). The exit
                # codes 11 and 12 tell the two steps apart.
                python = Path(sys.executable).with_name('python.exe')
                cython = [python, '-m', 'cython', '-3', '-I', folder, '-o', c_file, pyx]
                script = ['@echo off', 'chcp 65001 >nul']
                for code, command in enumerate([cython, gcc], 11):
                    line = ' '.join('"' + str(a).replace('%', '%%') + '"' for a in command)
                    script += [line, f'if errorlevel 1 exit /b {code}']
                try:
                    bat.write_text('\n'.join(script) + '\n', encoding='utf-8')
                    done = subprocess.run(
                        ['cmd.exe', '/c', os.curdir + os.sep + bat.name], cwd=out,
                        capture_output=True, text=True, encoding='utf-8',
                        errors='replace', creationflags=priv.NO_WINDOW)
                finally:
                    bat.unlink(missing_ok=True)
                if done.returncode:
                    step = {11: 'Cython', 12: 'gcc'}.get(done.returncode, 'cmd')
                    raise CompileError(
                        f'{name}.pyx: {step} failed (code {done.returncode})\n'
                        f'{(done.stderr or done.stdout)[-800:]}')
                c_file.unlink()
                importlib.invalidate_caches()   # the finders listed the folder before
            module = importlib.import_module(name)
            file = vars(module).get('__file__')
            there = Path(file).parent if file else folder
            there = there.parent if there.name == priv.PYXOUT else there
            if os.path.normcase(str(there)) != os.path.normcase(str(folder)):
                print(f'  use: WARNING {name} was imported from {there}, not from '
                      f'{folder}, which was found.')
